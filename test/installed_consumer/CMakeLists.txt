cmake_minimum_required(VERSION 3.16)
project(MotorControlRSReadConsumer LANGUAGES CXX)
file(TO_CMAKE_PATH "${MOTORCONTROLRS_SOURCE_DIR}" MOTORCONTROLRS_SOURCE_DIR)
file(TO_CMAKE_PATH "${MOTORCONTROLRS_HEADERS_DIR}" MOTORCONTROLRS_HEADERS_DIR)
if(MOTORCONTROLRS_SOURCE_DIR)
    set(BUILD_TESTING OFF CACHE BOOL "Do not build package-private tests" FORCE)
    add_subdirectory("${MOTORCONTROLRS_SOURCE_DIR}" core)
elseif(MOTORCONTROLRS_EXPECTED_VERSION)
    find_package(MotorControlRS ${MOTORCONTROLRS_EXPECTED_VERSION} EXACT CONFIG REQUIRED)
else()
    find_package(MotorControlRS CONFIG REQUIRED)
endif()

function(strict_consumer target standard)
    target_link_libraries(${target} PRIVATE MotorControlRS::MotorControlRS)
    set_target_properties(${target} PROPERTIES
        CXX_STANDARD ${standard} CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -pedantic-errors -Werror)
    endif()
    if(standard EQUAL 17)
        if(MSVC)
            target_compile_options(${target} PRIVATE /GR-)
        else()
            target_compile_options(${target} PRIVATE -fno-rtti)
        endif()
    endif()
endfunction()

add_executable(read_consumer main.cpp)
strict_consumer(read_consumer 11)
add_executable(embedded_consumer main.cpp)
strict_consumer(embedded_consumer 17)
add_executable(codec_consumer codec.cpp)
strict_consumer(codec_consumer 11)

# This path belongs to the staged source or installed package, never the checkout.
if(NOT IS_DIRECTORY "${MOTORCONTROLRS_HEADERS_DIR}/MotorControlRS")
    message(FATAL_ERROR "Set MOTORCONTROLRS_HEADERS_DIR to the package's public include directory")
endif()
file(GLOB_RECURSE public_headers CONFIGURE_DEPENDS RELATIVE "${MOTORCONTROLRS_HEADERS_DIR}"
    "${MOTORCONTROLRS_HEADERS_DIR}/MotorControlRS/*.h")
foreach(standard 11 17)
    set(header_sources)
    foreach(header IN LISTS public_headers)
        string(MAKE_C_IDENTIFIER "${header}" header_id)
        set(source "${CMAKE_CURRENT_BINARY_DIR}/headers_${standard}/${header_id}.cpp")
        file(WRITE "${source}" "#include <${header}>\n")
        list(APPEND header_sources "${source}")
    endforeach()
    add_library(public_headers_${standard} OBJECT ${header_sources})
    strict_consumer(public_headers_${standard} ${standard})
endforeach()

enable_testing()
add_test(NAME installed_read_consumer COMMAND read_consumer)
add_test(NAME embedded_consumer COMMAND embedded_consumer)
add_test(NAME codec_consumer COMMAND codec_consumer)

cmake_minimum_required(VERSION 3.16)
project(MotorControlRSReadConsumer LANGUAGES CXX)
find_package(MotorControlRS CONFIG REQUIRED)
add_executable(read_consumer main.cpp)
target_link_libraries(read_consumer PRIVATE MotorControlRS::MotorControlRS)
set_target_properties(read_consumer PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(read_consumer PRIVATE /W4 /WX)
else()
    target_compile_options(read_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME installed_read_consumer COMMAND read_consumer)
